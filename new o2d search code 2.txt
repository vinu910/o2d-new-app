// =====================================================================
// CACHE HELPERS — localStorage mein data store karo
// =====================================================================
var CACHE_PREFIX = 'ufsm_cache_';
var CACHE_TS_PREFIX = 'ufsm_ts_';

function getCacheKey(sheetId, sheetName) {
  return CACHE_PREFIX + sheetId + '_' + sheetName;
}

function saveToCache(sheetId, sheetName, parsed) {
  try {
    var key = getCacheKey(sheetId, sheetName);
    localStorage.setItem(key, JSON.stringify(parsed));
    localStorage.setItem(CACHE_TS_PREFIX + sheetId + '_' + sheetName, Date.now().toString());
  } catch(e) { /* storage full ho toh ignore */ }
}

function loadFromCache(sheetId, sheetName) {
  try {
    var key = getCacheKey(sheetId, sheetName);
    var raw = localStorage.getItem(key);
    if (!raw) return null;
    return JSON.parse(raw);
  } catch(e) { return null; }
}

function getCacheAge(sheetId, sheetName) {
  try {
    var ts = localStorage.getItem(CACHE_TS_PREFIX + sheetId + '_' + sheetName);
    if (!ts) return Infinity;
    return (Date.now() - parseInt(ts, 10)) / 1000; // seconds
  } catch(e) { return Infinity; }
}

    } catch(e) { onError(e); }
  };

  _jsonpTimer = setTimeout(function() {
    cleanup();
    onError(new Error('Timeout — 15s mein load nahi hua'));
  }, 15000);

// =====================================================================
// PARSE GVIZ
// =====================================================================
function parseGvizTable(table) {
  if (!table || !table.cols) throw new Error('No table data');
  var colLetters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
  var cols = [];
  for (var c = 0; c < table.cols.length; c++) {
    var col = table.cols[c];
    var label = (col.label && col.label.trim()) ? col.label.trim() :
                (col.id   && col.id.trim()   ) ? col.id.trim()    :
                (colLetters[c] || ('Col' + (c+1)));
    cols.push(label);
  }
  var rows = [];
  for (var r = 0; r < table.rows.length; r++) {
    var cells = table.rows[r].c;
    var obj = {}; var allEmpty = true;
    for (var ci = 0; ci < cols.length; ci++) {
      var cell = (cells && cells[ci]) ? cells[ci] : null;
      var val = '';
      if (cell && cell.v !== null && cell.v !== undefined) {
        val = (cell.f !== undefined && cell.f !== null) ? String(cell.f) : String(cell.v);
        if (val !== '') allEmpty = false;
      }
      obj[cols[ci]] = val;
    }
    if (!allEmpty) rows.push(obj);
  }
  return {cols: cols, rows: rows};
}

// =====================================================================
// PROCESS PARSED DATA — allData/visibleCols build karo
// =====================================================================
function processData(parsed) {
  headers = parsed.cols;
  visibleCols = [];
  for (var i = 0; i < headers.length; i++) {
    if (HIDDEN[i]) continue;
    if (i === 14) {
      visibleCols.push({label:'📦 Item (E/G)', realIdx:VCOL_ITEM, vctype:'item'});
      visibleCols.push({label:'🔢 Qty (F↑)',   realIdx:VCOL_QTY,  vctype:'qty'});
    }
    visibleCols.push({label:headers[i], realIdx:i});
  }
  var lastQty = '';
  allData = parsed.rows.map(function(row) {
    var eVal = (headers[4] && row[headers[4]]) ? String(row[headers[4]]).trim() : '';
    var gVal = (headers[6] && row[headers[6]]) ? String(row[headers[6]]).trim() : '';
    var fVal = (headers[5] && row[headers[5]]) ? String(row[headers[5]]).trim() : '';
    var effItem = eVal || gVal;
    var effSrc  = eVal ? 'E' : (gVal ? 'G' : '');
    var effQty = '', inherited = false;
    if (fVal !== '') { lastQty = fVal; effQty = fVal; }
    else             { effQty = lastQty; inherited = (lastQty !== ''); }
    var searchStr = '';
    for (var key in row) {
      if (Object.prototype.hasOwnProperty.call(row, key)) searchStr += ' ' + row[key];
    }
    searchStr += ' ' + effItem + ' ' + effQty;
    return Object.assign({}, row, {
      _effItem:effItem, _effSrc:effSrc,
      _effQty:effQty,  _effInherited:inherited,
      _search:searchStr.toLowerCase()
    });
  });
  filteredData = allData.slice();
}

// =====================================================================
// LOAD SHEET — Stale-while-revalidate strategy
// 1. Cache data turant dikhao (instant)
// 2. Background mein fresh data fetch karo
// =====================================================================
function loadSheet() {
  var link      = document.getElementById('sheetLink').value.trim();
  var sheetName = document.getElementById('sheetName').value.trim();
  if (!link || !sheetName) { alert('Sheet Link aur Tab Name daalo'); return; }

  var sheetId = extractSheetId(link);

  // STEP 1: Cache se instant display
  var cached = loadFromCache(sheetId, sheetName);
  if (cached && cached.rows && cached.rows.length) {
    var ageSeconds = getCacheAge(sheetId, sheetName);
    var ageStr = ageSeconds < 60 ? Math.round(ageSeconds) + 's' :
                 ageSeconds < 3600 ? Math.round(ageSeconds/60) + 'm' :
                 Math.round(ageSeconds/3600) + 'h';

    processData(cached);
    document.getElementById('thead').removeAttribute('data-loaded');
    renderTable(filteredData, false); // no animation for stale
    buildFilters();
    updateStats(filteredData);
    setBadge('⚡ Cache');
    setStatus('Cache se: ' + cached.rows.length + ' rows (age: ' + ageStr + ') — Fresh data aa raha hai...');

    // Stale banner dikhao
    document.getElementById('staleBanner').classList.add('visible');

    // Progress bar show karo
    showProgress();
  } else {
    // Koi cache nahi — normal loading state
    setBadge('Loading...');
    setStatus('Data fetch ho raha hai...');
    document.getElementById('tbody').innerHTML = '<tr><td class="loading-msg">⚡ Loading...</td></tr>';
    document.getElementById('thead').innerHTML = '';
    document.getElementById('thead').removeAttribute('data-loaded');
    showProgress();
  }

  // STEP 2: Background mein fresh data fetch
  localStorage.setItem('lastSheet', JSON.stringify({link: link, name: sheetName}));

  loadGvizJsonp(sheetId, sheetName,
    function(parsed) {
      hideProgress();
      document.getElementById('staleBanner').classList.remove('visible');

      if (!parsed.rows || !parsed.rows.length) {
        setBadge('Error');
        setStatus('Data nahi mila — Tab name check karo (case-sensitive)');
        document.getElementById('tbody').innerHTML =
          '<tr><td class="error-msg">Koi row nahi aayi.<br>Tab ka naam exactly wahi likho jo Google Sheet mein hai.</td></tr>';
        return;
      }

      // Cache mein save karo
      saveToCache(sheetId, sheetName, parsed);

      // Process & render fresh data
      processData(parsed);
      document.getElementById('thead').removeAttribute('data-loaded');
      renderTable(filteredData, true); // with animation = fresh
      buildFilters();
      updateStats(filteredData);
      setBadge('✅ Live');
      setStatus('✅ Fresh: ' + parsed.rows.length + ' rows · Col M,N,P hidden · Item(E/G) & Qty(F↑) added');
      showToast('✅ Fresh data loaded!', 'ok');
    },
    function(err) {
      hideProgress();
      document.getElementById('staleBanner').classList.remove('visible');

      // Agar cache data tha toh error sirf toast mein dikhao
      if (allData.length > 0) {
        setBadge('⚠️ Offline');
        setStatus('⚠️ Fresh data nahi aaya (' + err.message + ') — Cache data dikh raha hai');
        showToast('⚠️ Cache data dikh raha hai', 'warn');
      } else {
        setBadge('Error');
        setStatus('❌ ' + err.message);
        document.getElementById('tbody').innerHTML =
          '<tr><td class="error-msg">❌ ' + err.message + '<br><br>' +
          '<b>Checklist:</b><br>' +
          '1. Share → "Anyone with the link" → Viewer<br>' +
          '2. Tab name exactly wahi (capital/small matter karta hai)<br>' +
          '3. Sheet URL sahi paste karo</td></tr>';
      }
      console.error(err);
    }
  );
}

function renderTable(data, animate) {
  var thead = document.getElementById('thead');
  var tbody = document.getElementById('tbody');
  var table = document.getElementById('mainTable');

  if (!data.length) {
    tbody.innerHTML = '<tr><td class="loading-msg" colspan="99">No matching data</td></tr>';
    return;
  }
  if (!thead.getAttribute('data-loaded')) {
    var hHtml = '<tr><th>#</th>';
    for (var vi = 0; vi < visibleCols.length; vi++) {
      var vc = visibleCols[vi];
      if      (vc.vctype === 'item') hHtml += '<th class="vcol">' + vc.label + '</th>';
      else if (vc.vctype === 'qty')  hHtml += '<th class="vcol-q">' + vc.label + '</th>';
      else                           hHtml += '<th>' + vc.label + '</th>';
    }
    hHtml += '</tr>';
    thead.innerHTML = hHtml;
    thead.setAttribute('data-loaded', '1');
  }

  // Animation class add/remove
  if (animate) { table.classList.add('fresh-load'); }
  else { table.classList.remove('fresh-load'); }

  var colQ = headers[16];
  var html = '';
  // ⚡ DocumentFragment virtual DOM — faster rendering
  for (var ri = 0; ri < data.length; ri++) {
    var row = data[ri];
    html += '<tr><td>' + (ri+1) + '</td>';
    for (var ci = 0; ci < visibleCols.length; ci++) {
      var col = visibleCols[ci];
      var idx = col.realIdx;
      if (idx === VCOL_ITEM) {
        var ival = row._effItem || '—';
        var isrc = row._effSrc  || '';
        html += '<td class="vcol-item" title="' + ival + '">' + ival + '<span class="src">' + isrc + '</span></td>';
        continue;
      }
      if (idx === VCOL_QTY) {
        var qval = row._effQty || '—';
        var qinh = row._effInherited;
        var qbadge = qinh ? '↑prev' : 'F';
        var qcls   = qinh ? 'sf si' : 'sf';
        html += '<td class="vcol-qty' + (qinh?' inherited':'') + '" title="' +
          (qinh?'Upar wali row se aaya':'Col F se') + '">' +
          qval + '<span class="src ' + qcls + '">' + qbadge + '</span></td>';
        continue;
      }
      var hname = headers[idx];
      var raw   = (hname && row[hname] !== undefined) ? row[hname] : '';
      var display = raw;
      if (hname === colQ) {
        var sv = String(raw).toLowerCase();
        var sc = 'pending';
        if (sv.indexOf('deliver') !== -1) sc = 'delivered';
        if (sv.indexOf('partial') !== -1) sc = 'partial';
        if (sv.indexOf('cancel')  !== -1) sc = 'cancelled';
        display = '<span class="status-pill ' + sc + '">' + raw + '</span>';
      }
      if (idx === 7)  display = '<span class="qty-green">' + raw + '</span>';
      if (idx === 10) display = '<span class="qty-red">'   + raw + '</span>';
      html += '<td title="' + String(raw).replace(/"/g,'&quot;') + '">' + display + '</td>';
    }
    html += '</tr>';
  }
  tbody.innerHTML = html;

  // Remove animation class after done (reset for next time)
  if (animate) {
    setTimeout(function() { table.classList.remove('fresh-load'); }, 500);
  }
}

var FILTER_CFG = [
  {id:'filterB', colIdx:1,  ph:'Cust PO'},
  {id:'filterD', colIdx:3,  ph:'Company'},
  {id:'filterG', colIdx:6,  ph:'Send Item'},
  {id:'filterK', colIdx:10, ph:'Pending Item'},
  {id:'filterQ', colIdx:16, ph:'Status'}
];

function buildFilters()      { refreshDropdowns(allData); }
function refreshDropdowns(base) {
  for (var fi = 0; fi < FILTER_CFG.length; fi++) {
    var cfg = FILTER_CFG[fi];
    var sel = document.getElementById(cfg.id);
    var col = headers[cfg.colIdx];
    if (!col) continue;
    var cur = sel.value;
    var seen = {}, vals = [];
    for (var ri = 0; ri < base.length; ri++) {
      var v = base[ri][col];
      if (v && String(v).trim() !== '' && !seen[v]) { seen[v]=true; vals.push(v); }
    }
    vals.sort();
    var html = '<option value="">' + cfg.ph + '</option>';
    for (var vi = 0; vi < vals.length; vi++) html += '<option value="' + vals[vi] + '">' + vals[vi] + '</option>';
    sel.innerHTML = html;
    if (cur && seen[cur]) sel.value = cur;
    applyFilterStyle(cfg.id);
  }
}

function doSearch() {
  var text = document.getElementById('search').value.toLowerCase().trim();
  var bv=document.getElementById('filterB').value, dv=document.getElementById('filterD').value;
  var gv=document.getElementById('filterG').value, kv=document.getElementById('filterK').value;
  var qv=document.getElementById('filterQ').value;
  var colB=headers[1], colD=headers[3], colG=headers[6], colK=headers[10], colQ=headers[16];
  filteredData = [];
  for (var i = 0; i < allData.length; i++) {
    var row = allData[i];
    if (text && row._search.indexOf(text)===-1) continue;
    if (bv && row[colB]!==bv) continue;
    if (dv && row[colD]!==dv) continue;
    if (gv && row[colG]!==gv) continue;
    if (kv && row[colK]!==kv) continue;
    if (qv && row[colQ]!==qv) continue;
    filteredData.push(row);
  }
  renderTable(filteredData, false);
  updateStats(filteredData);
  refreshDropdowns(filteredData);
  buildSuggestions(text);
}

function buildSuggestions(text) {
  var box = document.getElementById('suggestions');
  if (!text) { box.style.display='none'; return; }
  var matches = [];
  outer:
  for (var i = 0; i < allData.length; i++) {
    var row = allData[i];
    for (var key in row) {
      if (!Object.prototype.hasOwnProperty.call(row,key)) continue;
      if (key.charAt(0)==='_') continue;
      var val = String(row[key]);
      if (val.toLowerCase().indexOf(text)!==-1 && val.trim()!=='') {
        var dup = false;
        for (var m=0;m<matches.length;m++) { if (matches[m]===val) { dup=true; break; } }
        if (!dup) { matches.push(val); if (matches.length>=10) break outer; }
      }
    }
  }
  var html = '';
  for (var mi = 0; mi < matches.length; mi++) {
    var escaped = matches[mi].replace(/\\/g,'\\\\').replace(/'/g,"\\'");
    html += '<div onclick="pickSuggestion(\'' + escaped + '\')">' + matches[mi] + '</div>';
  }
  box.innerHTML = html;
  box.style.display = matches.length ? 'block' : 'none';
}

function pickSuggestion(v) {
  document.getElementById('search').value = v;
  document.getElementById('suggestions').style.display = 'none';
  doSearch();
}

document.addEventListener('click', function(e) {
  if (!e.target.closest('.searchbox')) document.getElementById('suggestions').style.display='none';
});

function updateStats(data) {
  document.getElementById('sTotalRows').textContent = allData.length.toLocaleString();
  document.getElementById('sFiltered').textContent  = data.length.toLocaleString();
  var colF=headers[5], colH=headers[7], colL=headers[11];
  var tq=0, sq=0, pq=0;
  for (var i=0; i<data.length; i++) {
    var r=data[i];
    tq += parseFloat(String(r[colF]||'0').replace(/,/g,''))||0;
    sq += parseFloat(String(r[colH]||'0').replace(/,/g,''))||0;
    pq += parseFloat(String(r[colL]||'0').replace(/,/g,''))||0;
  }
  document.getElementById('sTotalQty').textContent   = tq.toLocaleString();
  document.getElementById('sSendQty').textContent    = sq.toLocaleString();
  document.getElementById('sPendingQty').textContent = pq.toLocaleString();
}
